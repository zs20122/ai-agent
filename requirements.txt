# 安装：python -m pip install -r requirements.txt

#             版本由 pip 解析锁定后钉住，安装前已用 pip install --dry-run 确认不会改动上面的 langchain/langgraph。
# 2026-10-01：会话记忆支持 SQLite 持久化，新增 langgraph-checkpoint-sqlite（会带入 aiosqlite / sqlite-vec）；

# ---------- LLM 调用与 Agent 编排 ----------
langchain==1.4.3            # LangChain 主包：Prompt / Tool / Agent 抽象
langchain-openai==1.6.6     # OpenAI 兼容模型接入（ChatOpenAI、Embeddings）
langgraph==1.2.12           # 有状态多节点 Agent 编排（StateGraph、checkpointer）

# ---------- Web 服务 ----------
fastapi==0.142.1            # HTTP API 框架（异步、自动 OpenAPI 文档）
uvicorn==0.54.0             # ASGI 服务器，启动 fastapi 应用

# ---------- 配置与数据校验 ----------
python-dotenv==1.2.3        # 从 .env 读取密钥/配置到环境变量
pydantic==2.13.5            # 请求/响应与配置的数据校验（v2 API）

# ---------- Web 前端（Streamlit 聊天界面，见 frontend/app.py） ----------
streamlit==1.64.0           # 中文气泡聊天界面（st.chat_input / st.chat_message）
requests==2.34.2            # 前端调用后端 /api/v1/chat（上游已间接依赖，这里显式钉住，避免漂移）

# ---------- RAG（检索增强：本地向量化 + 向量检索） ----------
langchain-community==0.4.2      # 文档加载 / 文本切分 / 向量库适配（FAISS 等）的统一入口
faiss-cpu==1.15.1               # 本地向量索引（CPU 版，无需 GPU；Windows 提供 cp310 轮子）
sentence-transformers==6.1.0    # 本地 Embedding 模型（把文本转成向量，无需调用云端接口）

# ---------- 会话记忆持久化（CHECKPOINT_BACKEND=sqlite，见 app/memory/checkpointer.py） ----------
langgraph-checkpoint-sqlite==3.1.1  # AsyncSqliteSaver：会话历史落盘到 SQLITE_DB_PATH，重启后可续聊

# 说明：以下包由上面的依赖自动安装，代码里直接 import，无需（也不应）在此重复钉版本，
#      否则容易出现与上游约束冲突：
#   langchain_core        -> 由 langchain / langgraph / langchain-openai 共同约束（>=1.4.7,<2.0.0）
#                            项目直接使用：messages（AIMessage/HumanMessage/SystemMessage/ToolMessage）、tools（@tool/BaseTool）
#   langgraph_checkpoint  -> 由 langgraph 提供，提供 langgraph.checkpoint.memory.InMemorySaver
#   langgraph_prebuilt    -> 由 langgraph 提供（项目当前未使用，自研工具执行节点）
#   openai / httpx / jsonpatch / requests-toolbelt -> 由 langchain-openai 提供
#   langchain-classic / langchain-text-splitters / pydantic-settings / SQLAlchemy / aiohttp
#                             -> 由 langchain-community 提供（文档切分、向量库适配都在其中）
#   aiosqlite / sqlite-vec   -> 由 langgraph-checkpoint-sqlite 提供（SQLite 异步驱动与向量扩展）
#   torch / transformers / huggingface-hub / tokenizers / scikit-learn / scipy
#                             -> 由 sentence-transformers 提供，是本次新增里体积最大的一项（CPU 版 torch）；
#                                首次实际使用时还会联网下载 Embedding 模型权重，离线环境请提前缓存或指定本地模型目录
#
# 注意包名与 import 名的差异：langchain-openai 的 import 名是 langchain_openai。
# 会话记忆默认走内存；要落盘请把 .env 的 CHECKPOINT_BACKEND 改为 sqlite（依赖已在本清单中）。
# 若改用 Postgres 等其它后端，请自行追加对应包（如 langgraph-checkpoint-postgres）
# 并同步修改 app/memory/checkpointer.py。
